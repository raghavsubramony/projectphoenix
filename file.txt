# ✅ 2. Core idea: slider-crank + thermodynamics

Cylinder volume changes with crank angle:

$$
V(\theta) = V_c + \frac{\pi D^2}{4} \cdot x(\theta)
$$

Where piston position:

$$
x(\theta) = R (1 - \cos\theta) + L \left(1 - \sqrt{1 - (R/L)^2 \sin^2\theta}\right)
$$

# ✅ 3. Example: combustion cycle simulation in Python

This example simulates:

* Cylinder volume
* Simple combustion (heat addition)
* P–V diagram
* Optional piston animation

***

## 🔧 Python Code

```python
import numpy as np
import matplotlib.pyplot as plt

# Engine parameters
bore = 0.086        # meters
stroke = 0.086      # meters
con_rod = 0.143     # meters
cr = 10             # compression ratio

R = stroke / 2
L = con_rod

# Derived values
Vd = np.pi * (bore**2 / 4) * stroke
Vc = Vd / (cr - 1)

gamma = 1.4   # air
P0 = 101325   # Pa
T0 = 300      # K

theta = np.linspace(0, 2*np.pi, 500)

def piston_position(theta):
    return R * (1 - np.cos(theta)) + \
           L * (1 - np.sqrt(1 - (R/L)**2 * np.sin(theta)**2))

def volume(theta):
    return Vc + np.pi * (bore**2 / 4) * piston_position(theta)

V = volume(theta)

# Simple compression-expansion (isentropic)
P = P0 * (V[0] / V)**gamma

# Add simple combustion spike near TDC
combustion_center = np.pi
combustion_width = 0.2
heat_release = 5e6 * np.exp(-((theta - combustion_center)/combustion_width)**2)

P_comb = P + heat_release

# Plot PV diagram
plt.figure(figsize=(8,5))
plt.plot(V, P_comb, label="With Combustion")
plt.plot(V, P, '--', label="No Combustion")
plt.xlabel("Volume (m³)")
plt.ylabel("Pressure (Pa)")
plt.title("Cylinder Combustion Cycle (Simplified)")
plt.legend()
plt.grid()
plt.show()
```

# ✅ 3. Full Python Code

## 🔥 Combustion + Engine Simulation

```python
import cantera as ct
import numpy as np
import pyvista as pv

# --------------------------
# Engine parameters
# --------------------------
bore = 0.086
stroke = 0.086
con_rod = 0.143
cr = 10

R = stroke / 2
L = con_rod

Vd = np.pi * (bore**2 / 4) * stroke
Vc = Vd / (cr - 1)

rpm = 3000
omega = 2 * np.pi * rpm / 60

# Fuel injection timing
theta_inj_start = np.pi - 0.2   # before TDC
theta_inj_end   = np.pi + 0.2

# --------------------------
# Geometry functions
# --------------------------
def piston_position(theta):
    return R*(1 - np.cos(theta)) + \
           L*(1 - np.sqrt(1 - (R/L)**2 * np.sin(theta)**2))

def volume(theta):
    return Vc + (np.pi * bore**2 / 4) * piston_position(theta)

# --------------------------
# Cantera setup
# --------------------------
gas = ct.Solution('gri30.yaml')

gas.TPX = 300, ct.one_atm, 'O2:1, N2:3.76'
reactor = ct.IdealGasReactor(gas)

sim = ct.ReactorNet([reactor])

# --------------------------
# Simulation loop
# --------------------------
theta_vals = np.linspace(0, 2*np.pi, 400)
time = 0

pressure = []
temperature = []

for i, theta in enumerate(theta_vals):

    V = volume(theta)
    reactor.volume = V

    # Fuel injection (simple pulse)
    if theta_inj_start < theta < theta_inj_end:
        gas.TPX = reactor.T, reactor.thermo.P, 'CH4:0.1, O2:1, N2:3.76'
        reactor.syncState()

    # Advance simulation
    dt = 1 / (rpm * 10)
    time += dt
    sim.advance(time)

    pressure.append(reactor.thermo.P)
    temperature.append(reactor.T)

# --------------------------
# Plot results
# --------------------------
import matplotlib.pyplot as plt

plt.plot(theta_vals, pressure)
plt.xlabel("Crank Angle (rad)")
plt.ylabel("Pressure (Pa)")
plt.title("Pressure vs Crank Angle")
plt.grid()
plt.show()
```

# ✅ 4. 3D Piston-Cylinder Animation (PyVista)

```python
# Create 3D scene
plotter = pv.Plotter()

# Cylinder
cylinder = pv.Cylinder(radius=bore/2, height=stroke)
plotter.add_mesh(cylinder, opacity=0.2)

# Piston
piston = pv.Cylinder(radius=bore/2, height=0.02)
actor = plotter.add_mesh(piston, color='red')

# Animate
for theta in theta_vals:

    z = piston_position(theta)

    piston.center = (0, 0, z - stroke/2)
    actor.SetPosition(0, 0, z - stroke/2)

    plotter.render()

plotter.show()
```
